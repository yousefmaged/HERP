import kernel from '../../app/core/kernel.js'; const moduleId = 'knowledge'; const api = { async init() { await kernel.execute(` CREATE TABLE IF NOT EXISTS pages ( id TEXT PRIMARY KEY, title TEXT NOT NULL, content TEXT, created_at INTEGER ) `); console.log('[Knowledge] جاهز'); }, async createPage(title, content) { const id = `page_${Date.now()}`; await kernel.execute(`INSERT INTO pages (id, title, content, created_at) VALUES (?, ?, ?, ?)`, [id, title, content, Date.now()]); return id; }, async listPages() { return kernel.query(`SELECT * FROM pages ORDER BY created_at DESC`); }, async getPage(id) { const res = await kernel.query(`SELECT * FROM pages WHERE id = ?`, [id]); return res[0] || null; } }; kernel.registerModule(moduleId, api);
