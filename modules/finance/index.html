import kernel from '../../app/core/kernel.js'; const moduleId = 'finance'; const api = { async init() { await kernel.execute(` CREATE TABLE IF NOT EXISTS invoices ( id TEXT PRIMARY KEY, title TEXT NOT NULL, amount REAL, status TEXT DEFAULT 'pending', created_at INTEGER ) `); console.log('[Finance] جاهز'); }, async createInvoice(title, amount) { const id = `inv_${Date.now()}`; await kernel.execute(`INSERT INTO invoices (id, title, amount, created_at) VALUES (?, ?, ?, ?)`, [id, title, amount, Date.now()]); return id; }, async listInvoices() { return kernel.query(`SELECT * FROM invoices ORDER BY created_at DESC`); } }; kernel.registerModule(moduleId, api);
